title = "COUNT(*) 与 COUNT(列) 的计数范围"
keywords = "count aggregate group by null 计数 汇总 聚合 缺少匹配 LEFT JOIN"
source = "https://www.sqlite.org/lang_aggfunc.html#count"
applicable = "计算非空值数量，或在左连接后统计真实匹配记录。"
dataset_id = "retail"
wrong_sql = "SELECT c.id, COUNT(*) FROM customers c LEFT JOIN orders o ON c.id=o.customer_id GROUP BY c.id ORDER BY c.id;"
example_sql = "SELECT c.id, COUNT(o.id) FROM customers c LEFT JOIN orders o ON c.id=o.customer_id GROUP BY c.id ORDER BY c.id;"
+++
COUNT(*) 统计组内全部行，包括左连接生成的补位行。COUNT(表达式) 统计表达式有非 NULL 值的行。对匹配表的非空主键计数可得到实际匹配数。
